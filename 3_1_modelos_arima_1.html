# Function for unit root tests with detailed interpretation, including Random Walk
def unit_root_tests(series, name):
    print(f"\nUnit Root Tests for {name}:")

    # ADF Test
    adf_result = adfuller(series)
    print("ADF Test:")
    print(f'ADF Statistic: {adf_result[0]:.4f}')
    print(f'p-value: {adf_result[1]:.4f}')
    print(f'Critical Values: {adf_result[4]}')
    print("Interpretation:")
    if adf_result[1] < 0.05:
        print(f"  - p-value < 0.05: Reject null hypothesis - {name} is stationary")
        print(f"    Implication: {name} does not have a unit root and is suitable for modeling without differencing.")
        print(f"    Random Walk: {name} does NOT follow a Random Walk (since it is stationary).")
    else:
        print(f"  - p-value >= 0.05: Fail to reject null - {name} may be non-stationary")
        print(f"    Implication: {name} may have a unit root; consider differencing or further analysis.")
        print(f"    Random Walk: {name} may follow a Random Walk (since it has a unit root).")

    # KPSS Test
    kpss_result = kpss(series)
    print("\nKPSS Test:")
    print(f'KPSS Statistic: {kpss_result[0]:.4f}')
    print(f'p-value: {kpss_result[1]:.4f}')
    print(f'Critical Values: {kpss_result[3]}')
    print("Interpretation:")
    if kpss_result[1] < 0.05:
        print(f"  - p-value < 0.05: Reject null hypothesis - {name} is non-stationary")
        print(f"    Implication: {name} may require differencing to achieve stationarity for modeling.")
    else:
        print(f"  - p-value >= 0.05: Fail to reject null - {name} may be stationary")
        print(f"    Implication: {name} appears stationary and may not need differencing.")

    # Combined Interpretation
    print("\nCombined Interpretation:")
    if adf_result[1] < 0.05 and kpss_result[1] >= 0.05:
        print(f"  - Both tests agree: {name} is stationary. Suitable for modeling as-is.")
        print(f"    Random Walk Conclusion: {name} does NOT follow a Random Walk.")
    elif adf_result[1] >= 0.05 and kpss_result[1] < 0.05:
        print(f"  - Both tests agree: {name} is non-stationary. Differencing or transformation may be required.")
        print(f"    Random Walk Conclusion: {name} likely follows a Random Walk.")
    else:
        print(f"  - Tests disagree: Mixed evidence on stationarity for {name}. Consider further analysis (e.g., differencing, trend analysis).")
        print(f"    Random Walk Conclusion: Uncertain - further analysis needed to determine if {name} follows a Random Walk.")

# Perform unit root tests on original series

# Cointegration test with detailed interpretation
def cointegration_test(df):
    result = coint_johansen(df, det_order=0, k_ar_diff=1)
    print("\nJohansen Cointegration Test (JNJ, XOM, NVDA):")
    print(f"Trace statistic: {result.lr1}")
    print(f"Critical values (90%, 95%, 99%): {result.cvt}")
    print("Interpretation:")
    for i in range(len(result.lr1)):
        if result.lr1[i] > result.cvt[i, 1]:
            print(f"  - r = {i}: Cointegration exists at 95% confidence level")
            print(f"    Trace statistic ({result.lr1[i]:.2f}) > 95% critical value ({result.cvt[i, 1]:.2f})")
        else:
            print(f"  - r = {i}: No cointegration at 95% confidence level")
            print(f"    Trace statistic ({result.lr1[i]:.2f}) <= 95% critical value ({result.cvt[i, 1]:.2f})")
    if result.lr1[0] > result.cvt[0, 1]:
        print("Conclusion: JNJ, XOM, and NVDA are cointegrated - they share a long-run equilibrium relationship.")
        print("Implication: These series move together over the long term; consider a Vector Error Correction Model (VECM) for forecasting.")
    else:
        print("Conclusion: No evidence of cointegration among JNJ, XOM, and NVDA.")
        print("Implication: These series do not share a long-run relationship; model them separately (e.g., using ARIMA).")

# Function to find best ARIMA model with interpretation
def find_best_arima(series, name, max_p=3, max_d=2, max_q=3):
    best_aic = float('inf')
    best_order = None

    for p in range(max_p + 1):
        for d in range(max_d + 1):
            for q in range(max_q + 1):
                try:
                    model = ARIMA(series, order=(p, d, q))
                    results = model.fit()
                    if results.aic < best_aic:
                        best_aic = results.aic
                        best_order = (p, d, q)
                except:
                    continue

    print(f"\nBest ARIMA model for {name}:")
    print(f"Order: {best_order}")
    print(f"AIC: {best_aic:.2f}")
    print("Interpretation:")
    print(f"  - p={best_order[0]}: {best_order[0]} autoregressive term(s)")
    print(f"  - d={best_order[1]}: {best_order[1]} difference(s) needed for stationarity")
    print(f"  - q={best_order[2]}: {best_order[2]} moving average term(s)")
    return best_order

# Detailed forecast plot with confidence intervals and interpretation
def plot_forecast(model, series, name, steps=30):
    # Generate forecast
    forecast_obj = model.get_forecast(steps=steps)
    forecast = forecast_obj.predicted_mean
    conf_int = forecast_obj.conf_int()

    # Transform to log10 scale for plotting
    series_log = np.log10(series)
    forecast_log = np.log10(forecast)
    conf_int_log = np.log10(conf_int)

    forecast_index = range(len(series), len(series) + steps)

    # Plot in log10 scale
    plt.figure(figsize=(12,6))
    plt.plot(series_log, label=f'{name} Historical (log10)', color='blue')
    plt.plot(forecast_index, forecast_log, label='Forecast (log10)', color='red')
    plt.fill_between(forecast_index,
                    conf_int_log.iloc[:, 0],
                    conf_int_log.iloc[:, 1],
                    color='pink',
                    alpha=0.3,
                    label='95% Confidence Interval (log10)')
    plt.title(f'{name} Price Forecast (log10 Scale)')
    plt.legend()
    plt.show()

    # Forecast interpretation in original scale
    last_value = series.iloc[-1]
    mean_forecast = forecast.mean()
    print(f"\nForecast Interpretation for {name} (Original Scale):")
    print(f"Last observed value: {last_value:.2f}")
    print(f"Average forecast value: {mean_forecast:.2f}")
    print(f"Forecast change: {mean_forecast - last_value:.2f}")
    if mean_forecast > last_value:
        print("Trend: Upward forecast trend")
    elif mean_forecast < last_value:
        print("Trend: Downward forecast trend")
    else:
        print("Trend: Flat forecast trend")
    print(f"95% CI range at period {steps} (Original Scale): [{conf_int.iloc[-1, 0]:.2f}, {conf_int.iloc[-1, 1]:.2f}]")

# Generate detailed forecast plots and interpretations
plot_forecast(jnj_model, jnj_close, "JNJ")
plot_forecast(xom_model, xom_close, "XOM")
plot_forecast(nvda_model, nvda_close, "NVDA")

        print(f"  - p-value >= 0.05: Fail to reject null - {name} may be stationary")
        print(f"    Implication: {name} appears stationary and may not need differencing.")

    # Combined Interpretation
    print("\nCombined Interpretation:")
    if adf_result[1] < 0.05 and kpss_result[1] >= 0.05:
        print(f"  - Both tests agree: {name} is stationary. Suitable for modeling as-is.")
        print(f"    Random Walk Conclusion: {name} does NOT follow a Random Walk.")
    elif adf_result[1] >= 0.05 and kpss_result[1] < 0.05:
        print(f"  - Both tests agree: {name} is non-stationary. Differencing or transformation may be required.")
        print(f"    Random Walk Conclusion: {name} likely follows a Random Walk.")
    else:
        print(f"  - Tests disagree: Mixed evidence on stationarity for {name}. Consider further analysis (e.g., differencing, trend analysis).")
        print(f"    Random Walk Conclusion: Uncertain - further analysis needed to determine if {name} follows a Random Walk.")

# Function to find best ARIMA model with interpretation
def find_best_arima(series, name, max_p=3, max_d=2, max_q=3):
    best_aic = float('inf')
    best_order = None

    for p in range(max_p + 1):
        for d in range(max_d + 1):
            for q in range(max_q + 1):
                try:
                    model = ARIMA(series, order=(p, d, q))
                    results = model.fit()
                    if results.aic < best_aic:
                        best_aic = results.aic
                        best_order = (p, d, q)
                except:
                    continue

    print(f"\nBest ARIMA model for {name}:")
    print(f"Order: {best_order}")
    print(f"AIC: {best_aic:.2f}")
    print("Interpretation:")
    print(f"  - p={best_order[0]}: {best_order[0]} autoregressive term(s)")
    print(f"  - d={best_order[1]}: {best_order[1]} difference(s) needed for stationarity")
    print(f"  - q={best_order[2]}: {best_order[2]} moving average term(s)")
    return best_order

# Detailed forecast plot with confidence intervals and interpretation
def plot_forecast(model, series, name, steps=30):
    # Generate forecast
    forecast_obj = model.get_forecast(steps=steps)
    forecast = forecast_obj.predicted_mean
    conf_int = forecast_obj.conf_int()

    # Transform to log10 scale for plotting
    series_log = np.log10(series)
    forecast_log = np.log10(forecast)
    conf_int_log = np.log10(conf_int)

    forecast_index = range(len(series), len(series) + steps)

    # Plot in log10 scale
    plt.figure(figsize=(12,6))
    plt.plot(series_log, label=f'{name} Historical (log10)', color='blue')
    plt.plot(forecast_index, forecast_log, label='Forecast (log10)', color='red')
    plt.fill_between(forecast_index,
                    conf_int_log.iloc[:, 0],
                    conf_int_log.iloc[:, 1],
                    color='pink',
                    alpha=0.3,
                    label='95% Confidence Interval (log10)')
    plt.title(f'{name} Price Forecast (log10 Scale)')
    plt.legend()
    plt.show()

    # Forecast interpretation in original scale
    last_value = series.iloc[-1]
    mean_forecast = forecast.mean()
    print(f"\nForecast Interpretation for {name} (Original Scale):")
    print(f"Last observed value: {last_value:.2f}")
    print(f"Average forecast value: {mean_forecast:.2f}")
    print(f"Forecast change: {mean_forecast - last_value:.2f}")
    if mean_forecast > last_value:
        print("Trend: Upward forecast trend")
    elif mean_forecast < last_value:
        print("Trend: Downward forecast trend")
    else:
        print("Trend: Flat forecast trend")
    print(f"95% CI range at period {steps} (Original Scale): [{conf_int.iloc[-1, 0]:.2f}, {conf_int.iloc[-1, 1]:.2f}]")

# Generate detailed forecast plots and interpretations
plot_forecast(jnj_model, jnj_close, "JNJ")
plot_forecast(xom_model, xom_close, "XOM")
plot_forecast(nvda_model, nvda_close, "NVDA")
